% CONFIGURACIÓN COMPARTIDA
\begin{frame}{Configuración de entrenamiento y callbacks}
    \small

    \begin{block}{Configuración compartida}
        \begin{itemize}
            \item Entrada: imágenes de $224 \times 224 \times 3$,
                  sin normalización.
            \item División: 90\,\% entrenamiento y 10\,\% validación,
                  con \texttt{random\_state=1}.
            \item Pérdida: entropía cruzada categórica.
            \item Recorte de gradientes: \texttt{clipnorm=1.0}.
        \end{itemize}
    \end{block}

    \textbf{Optimizadores}
    \begin{itemize}
        \item Adam: $\beta_1=0.9$, $\beta_2=0.999$,
              $\epsilon=10^{-8}$ y AMSGrad activado.
        \item SGD: momentum $=0.9$ y Nesterov activado.
    \end{itemize}

    \textbf{Callbacks comunes}
    \begin{itemize}
        \item \textbf{ModelCheckpoint}: guarda el mejor modelo.
        \item \textbf{ReduceLROnPlateau}: factor 0.1, paciencia 10.
        \item \textbf{EarlyStopping}: paciencia 40.
        \item \textbf{TerminateOnNaN}: detiene ante pérdida NaN.
    \end{itemize}

    \footnotesize
    Los tres primeros monitorizan \texttt{val\_accuracy}.
\end{frame}
